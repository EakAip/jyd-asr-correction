# 1. 将服务文件复制到 systemd 目录
sudo cp /opt/jyd01/wangruihua/AI_Tutor/jyd_intent.service /etc/systemd/system/

# 2. 重新加载 systemd 配置
sudo systemctl daemon-reload

# 3. 启用开机自启
sudo systemctl enable jyd_intent.service

# 4. 立即启动服务
sudo systemctl start jyd_intent.service

# 5. 查看服务状态
sudo systemctl status jyd_intent.service
